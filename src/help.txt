// The help text of forthling (see src/help.rs). One line per word:
//   NAME ( effect ) description ;; code => what the code prints
// under a heading [WORDLIST]. CI checks that every word has a line and that every example prints
// what it says. The words are grouped as in COMPLIANCE.md.
[FORTH]
// System variables
BASE ( -- a-addr ) Address of the number base used by . and by the text interpreter. ;; 255 16 BASE ! . DECIMAL => FF
STATE ( -- a-addr ) Address of the compile state: non-zero while a definition is being compiled. ;; STATE @ . => 0
>IN ( -- a-addr ) Address of the offset in the input buffer where parsing continues.
BLK ( -- a-addr ) Address of the number of the block being interpreted, 0 for the terminal or a file.
SCR ( -- a-addr ) Address of the number of the block LIST shows last.
// Stack
DUP ( x -- x x ) Copy the top item. ;; 5 DUP . . => 5 5
DROP ( x -- ) Remove the top item. ;; 1 2 DROP . => 1
OVER ( x1 x2 -- x1 x2 x1 ) Copy the second item to the top. ;; 1 2 OVER . . . => 1 2 1
ROT ( x1 x2 x3 -- x2 x3 x1 ) Rotate the third item to the top. ;; 1 2 3 ROT . . . => 1 3 2
SWAP ( x1 x2 -- x2 x1 ) Exchange the top two items. ;; 1 2 SWAP . . => 1 2
NIP ( x1 x2 -- x2 ) Remove the second item. ;; 1 2 NIP . => 2
TUCK ( x1 x2 -- x2 x1 x2 ) Copy the top item below the second. ;; 1 2 TUCK . . . => 2 1 2
-ROT ( x1 x2 x3 -- x3 x1 x2 ) Rotate the top item down to the third place. ;; 1 2 3 -ROT . . . => 2 1 3
?DUP ( x -- 0 | x x ) Copy the top item only if it is not zero. ;; 0 ?DUP DEPTH . => 1
PICK ( xu ... x0 u -- xu ... x0 xu ) Copy item u, counted from 0 at the top. ;; 7 8 9 2 PICK . => 7
ROLL ( xu ... x0 u -- xu-1 ... x0 xu ) Move item u to the top. ;; 7 8 9 2 ROLL . . . => 7 9 8
DEPTH ( -- n ) The number of items on the data stack. ;; 1 2 3 DEPTH . => 3
THIRD ( x1 x2 x3 -- x1 x2 x3 x1 ) Copy the third item to the top (Gforth). ;; 1 2 3 THIRD . => 1
FOURTH ( x1 x2 x3 x4 -- x1 x2 x3 x4 x1 ) Copy the fourth item to the top (Gforth). ;; 1 2 3 4 FOURTH . => 1
UNDER+ ( n1 n2 n3 -- n1+n3 n2 ) Add the top item to the third (Gforth). ;; 1 2 10 UNDER+ . . => 2 11
2DUP ( x1 x2 -- x1 x2 x1 x2 ) Copy the top pair. ;; 1 2 2DUP . . . . => 2 1 2 1
2DROP ( x1 x2 -- ) Remove the top pair. ;; 1 2 3 2DROP . => 1
2SWAP ( x1 x2 x3 x4 -- x3 x4 x1 x2 ) Exchange the top two pairs. ;; 1 2 3 4 2SWAP . . . . => 2 1 4 3
2OVER ( x1 x2 x3 x4 -- x1 x2 x3 x4 x1 x2 ) Copy the second pair to the top. ;; 1 2 3 4 2OVER . . => 2 1
2ROT ( x1 x2 x3 x4 x5 x6 -- x3 x4 x5 x6 x1 x2 ) Rotate the third pair to the top. ;; 1 2 3 4 5 6 2ROT . . => 2 1
2NIP ( x1 x2 x3 x4 -- x3 x4 ) Remove the second pair (Gforth). ;; 1 2 3 4 2NIP . . => 4 3
2TUCK ( x1 x2 x3 x4 -- x3 x4 x1 x2 x3 x4 ) Copy the top pair below the second pair (Gforth). ;; 1 2 3 4 2TUCK DEPTH . => 6
>R ( x -- ) ( R: -- x ) Move the top item to the return stack. Use it only inside a definition. ;; : T 5 >R R> ; T . => 5
R> ( -- x ) ( R: x -- ) Move the top of the return stack to the data stack. ;; : T 5 >R R> ; T . => 5
R@ ( -- x ) ( R: x -- x ) Copy the top of the return stack. ;; : T 5 >R R@ R> + ; T . => 10
RDROP ( R: x -- ) Drop the top of the return stack (Gforth).
2RDROP ( R: x1 x2 -- ) Drop the top two items of the return stack (Gforth).
RPICK ( u -- x ) ( R: ... -- ... ) Copy item u of the return stack, 0 being the top (Gforth).
2>R ( x1 x2 -- ) ( R: -- x1 x2 ) Move the top pair to the return stack. ;; : T 1 2 2>R 2R> ; T . . => 2 1
2R> ( -- x1 x2 ) ( R: x1 x2 -- ) Move a pair from the return stack to the data stack.
2R@ ( -- x1 x2 ) ( R: x1 x2 -- x1 x2 ) Copy the top pair of the return stack.
N>R ( x1 ... xn n -- ) ( R: -- x1 ... xn n ) Move n items and their count to the return stack.
NR> ( -- x1 ... xn n ) ( R: x1 ... xn n -- ) Move what N>R stored back to the data stack.
// Arithmetic
+ ( n1 n2 -- n3 ) Add. ;; 2 3 + . => 5
- ( n1 n2 -- n3 ) Subtract n2 from n1. ;; 7 2 - . => 5
* ( n1 n2 -- n3 ) Multiply. ;; 6 7 * . => 42
/ ( n1 n2 -- n3 ) Divide, rounding towards negative infinity (floored). ;; -7 2 / . => -4
MOD ( n1 n2 -- n3 ) The remainder of a floored division; it has the sign of n2. ;; -7 2 MOD . => 1
/MOD ( n1 n2 -- n3 n4 ) Floored division: remainder n3, quotient n4. ;; 7 2 /MOD . . => 3 1
*/ ( n1 n2 n3 -- n4 ) Multiply n1 by n2 in a double cell, then divide by n3 (floored). ;; 10 3 4 */ . => 7
*/MOD ( n1 n2 n3 -- n4 n5 ) Like */ but gives the remainder n4 and the quotient n5. ;; 10 3 4 */MOD . . => 7 2
1+ ( n1 -- n2 ) Add one. ;; 5 1+ . => 6
1- ( n1 -- n2 ) Subtract one. ;; 5 1- . => 4
2* ( x1 -- x2 ) Shift left one bit: multiply by two. ;; 21 2* . => 42
2/ ( x1 -- x2 ) Shift right one bit, keeping the sign: divide by two, rounding down. ;; -7 2/ . => -4
NEGATE ( n1 -- n2 ) Change the sign. ;; 5 NEGATE . => -5
ABS ( n1 -- n2 ) The absolute value. ;; -5 ABS . => 5
MIN ( n1 n2 -- n3 ) The smaller of two numbers. ;; 3 9 MIN . => 3
MAX ( n1 n2 -- n3 ) The larger of two numbers. ;; 3 9 MAX . => 9
UMIN ( u1 u2 -- u3 ) The smaller of two unsigned numbers (Gforth).
UMAX ( u1 u2 -- u3 ) The larger of two unsigned numbers (Gforth).
WITHIN ( n1|u1 n2|u2 n3|u3 -- flag ) True if n2 <= n1 < n3, with wrap-around. ;; 5 1 10 WITHIN . => -1
U/MOD ( u1 u2 -- u3 u4 ) Unsigned division: remainder u3, quotient u4 (Gforth). ;; 7 2 U/MOD . . => 3 1
UMOD ( u1 u2 -- u3 ) The remainder of an unsigned division (Gforth). ;; 7 2 UMOD . => 1
S>D ( n -- d ) Make a double-cell number from a single. ;; -5 S>D D. => -5
M* ( n1 n2 -- d ) Multiply two numbers into a double. ;; 100000000000 100000000000 M* D. => 10000000000000000000000
UM* ( u1 u2 -- ud ) Multiply two unsigned numbers into an unsigned double.
UM/MOD ( ud u1 -- u2 u3 ) Unsigned division of a double by a single: remainder u2, quotient u3.
FM/MOD ( d n1 -- n2 n3 ) Floored division of a double by a single: remainder n2, quotient n3.
SM/REM ( d n1 -- n2 n3 ) Symmetric division (towards zero) of a double by a single: remainder n2, quotient n3.
M*/ ( d1 n1 +n2 -- d2 ) Multiply d1 by n1 into a triple, divide by n2 (Forth-2012 double-number words).
M+ ( d1 n -- d2 ) Add a single to a double. ;; 5. 3 M+ D. => 8
LSHIFT ( x1 u -- x2 ) Shift left u bits; a shift of a cell or more gives 0. ;; 1 4 LSHIFT . => 16
RSHIFT ( x1 u -- x2 ) Shift right u bits without sign extension. ;; 256 4 RSHIFT . => 16
ARSHIFT ( x1 u -- x2 ) Shift right u bits, copying the sign bit (Gforth). ;; -16 2 ARSHIFT . => -4
ROL ( x1 u -- x2 ) Rotate left u bits (Gforth).
ROR ( x1 u -- x2 ) Rotate right u bits (Gforth).
CTZ ( x -- u ) The number of trailing zero bits; 64 for 0 (Gforth). ;; 8 CTZ . => 3
LOG2 ( x -- u ) The base-2 logarithm, rounded down: the position of the highest set bit (Gforth). ;; 1024 LOG2 . => 10
AND ( x1 x2 -- x3 ) Bitwise and. ;; 12 10 AND . => 8
OR ( x1 x2 -- x3 ) Bitwise or. ;; 12 10 OR . => 14
XOR ( x1 x2 -- x3 ) Bitwise exclusive or. ;; 12 10 XOR . => 6
INVERT ( x1 -- x2 ) Flip every bit. ;; 0 INVERT . => -1
// Comparison
= ( x1 x2 -- flag ) True if the two items are equal. ;; 3 3 = . => -1
<> ( x1 x2 -- flag ) True if the two items differ. ;; 3 4 <> . => -1
< ( n1 n2 -- flag ) True if n1 is less than n2. ;; 1 2 < . => -1
> ( n1 n2 -- flag ) True if n1 is greater than n2. ;; 1 2 > . => 0
<= ( n1 n2 -- flag ) True if n1 is less than or equal to n2 (Gforth). ;; 2 2 <= . => -1
>= ( n1 n2 -- flag ) True if n1 is greater than or equal to n2 (Gforth). ;; 1 2 >= . => 0
U< ( u1 u2 -- flag ) Unsigned less than. ;; -1 1 U< . => 0
U> ( u1 u2 -- flag ) Unsigned greater than. ;; -1 1 U> . => -1
U<= ( u1 u2 -- flag ) Unsigned less than or equal (Gforth).
U>= ( u1 u2 -- flag ) Unsigned greater than or equal (Gforth).
0= ( x -- flag ) True if the item is zero. ;; 0 0= . => -1
0<> ( x -- flag ) True if the item is not zero. ;; 5 0<> . => -1
0< ( n -- flag ) True if the number is negative. ;; -1 0< . => -1
0> ( n -- flag ) True if the number is positive. ;; 1 0> . => -1
0<= ( n -- flag ) True if the number is zero or negative (Gforth).
0>= ( n -- flag ) True if the number is zero or positive (Gforth).
TRUE ( -- true ) The flag true: all bits set, -1. ;; TRUE . => -1
FALSE ( -- false ) The flag false: 0. ;; FALSE . => 0
BL ( -- char ) The character code of a space, 32. ;; BL . => 32
// Double numbers
D+ ( d1 d2 -- d3 ) Add two doubles. ;; 5. 6. D+ D. => 11
D- ( d1 d2 -- d3 ) Subtract d2 from d1.
DNEGATE ( d1 -- d2 ) Change the sign of a double.
DABS ( d1 -- d2 ) The absolute value of a double.
DMAX ( d1 d2 -- d3 ) The larger of two doubles.
DMIN ( d1 d2 -- d3 ) The smaller of two doubles.
D2* ( d1 -- d2 ) Multiply a double by two.
D2/ ( d1 -- d2 ) Divide a double by two, rounding down.
D< ( d1 d2 -- flag ) True if d1 is less than d2.
D> ( d1 d2 -- flag ) True if d1 is greater than d2 (Gforth).
D= ( d1 d2 -- flag ) True if the doubles are equal.
D<> ( d1 d2 -- flag ) True if the doubles differ (Gforth).
DU< ( ud1 ud2 -- flag ) Unsigned less than for doubles.
D0< ( d -- flag ) True if the double is negative.
D0= ( d -- flag ) True if the double is zero.
D0<> ( d -- flag ) True if the double is not zero (Gforth).
D>S ( d -- n ) Make a single from a double, dropping the high cell.
2@ ( a-addr -- x1 x2 ) Fetch a pair; x2 is at the lower address. ;; CREATE P 5 , 6 , P 2@ . . => 5 6
2! ( x1 x2 a-addr -- ) Store a pair; x2 goes to the lower address.
2CONSTANT ( x1 x2 "name" -- ) Define a constant that gives a pair. ;; 1 2 2CONSTANT PAIR PAIR . . => 2 1
2VARIABLE ( "name" -- ) Define a variable of two cells.
2LITERAL ( x1 x2 -- ) Compile a pair as a literal. Use it inside a definition.
2VALUE ( x1 x2 "name" -- ) Define a value that holds a pair; TO changes it. ;; 1 2 2VALUE P P . . => 2 1
D. ( d -- ) Print a double and a space. ;; 123456789012345. D. => 123456789012345
D.R ( d n -- ) Print a double right-aligned in a field of n characters.
UD. ( ud -- ) Print an unsigned double and a space (Gforth).
UD.R ( ud n -- ) Print an unsigned double right-aligned in a field of n characters (Gforth).
// Memory
@ ( a-addr -- x ) Fetch the cell at the address. ;; VARIABLE V 9 V ! V @ . => 9
! ( x a-addr -- ) Store a cell at the address. ;; VARIABLE V 9 V ! V @ . => 9
+! ( n a-addr -- ) Add n to the cell at the address. ;; VARIABLE V 5 V ! 2 V +! V @ . => 7
C@ ( c-addr -- char ) Fetch a byte.
C! ( char c-addr -- ) Store a byte.
W@ ( addr -- u ) Fetch a 16-bit unsigned value (Gforth).
W! ( w addr -- ) Store the low 16 bits of a value (Gforth).
L@ ( addr -- u ) Fetch a 32-bit unsigned value (Gforth).
L! ( l addr -- ) Store the low 32 bits of a value (Gforth).
, ( x -- ) Reserve a cell in data space and store x in it. ;; CREATE T 7 , T @ . => 7
C, ( char -- ) Reserve a byte in data space and store it.
W, ( w -- ) Reserve two bytes in data space and store a 16-bit value (Gforth).
L, ( l -- ) Reserve four bytes in data space and store a 32-bit value (Gforth).
HERE ( -- addr ) The next free address in data space.
ALLOT ( n -- ) Reserve n bytes of data space; a negative n gives space back.
UNUSED ( -- u ) The number of bytes of data space that are still free.
PAD ( -- c-addr ) The address of a scratch area for text.
ALIGN ( -- ) Align HERE to a cell boundary.
ALIGNED ( addr -- a-addr ) Round an address up to a cell boundary.
CELLS ( n1 -- n2 ) The number of bytes in n1 cells (8 each). ;; 3 CELLS . => 24
CELL+ ( a-addr1 -- a-addr2 ) Add the size of a cell to an address. ;; 0 CELL+ . => 8
CELL- ( a-addr1 -- a-addr2 ) Subtract the size of a cell from an address (Gforth).
CHARS ( n1 -- n2 ) The number of bytes in n1 characters (1 each).
CHAR+ ( c-addr1 -- c-addr2 ) Add the size of a character to an address.
FILL ( c-addr u char -- ) Fill u bytes with the character. ;; PAD 3 65 FILL PAD 3 TYPE => AAA
ERASE ( addr u -- ) Fill u bytes with zeros.
BLANK ( c-addr u -- ) Fill u bytes with spaces.
MOVE ( addr1 addr2 u -- ) Copy u bytes; the areas may overlap.
CMOVE ( c-addr1 c-addr2 u -- ) Copy u bytes from the lowest address up.
CMOVE> ( c-addr1 c-addr2 u -- ) Copy u bytes from the highest address down.
BOUNDS ( addr u -- addr+u addr ) Turn a string into the limits of a DO loop (Gforth). ;; 100 3 BOUNDS . . => 100 103
BUFFER: ( u "name" -- ) Define a word that gives the address of u bytes of data space.
ALLOCATE ( u -- a-addr ior ) Allocate u bytes on the heap; ior is 0 on success. ;; 16 ALLOCATE THROW FREE THROW .S => <0>
FREE ( a-addr -- ior ) Give a block from ALLOCATE back to the heap.
RESIZE ( a-addr1 u -- a-addr2 ior ) Change the size of a block from ALLOCATE, keeping its contents.
VARIABLE ( "name" -- ) Define a variable of one cell. ;; VARIABLE X 4 X ! X @ . => 4
CONSTANT ( x "name" -- ) Define a constant. ;; 42 CONSTANT ANSWER ANSWER . => 42
VALUE ( x "name" -- ) Define a value. The name gives it; TO changes it. ;; 5 VALUE N 6 TO N N . => 6
UVALUE ( "name" -- ) Define a value that starts at 0 (Gforth).
? ( a-addr -- ) Print the cell at the address. ;; VARIABLE V 3 V ! V ? => 3
ON ( a-addr -- ) Store true in the cell (Gforth). ;; VARIABLE F F ON F @ . => -1
OFF ( a-addr -- ) Store false in the cell (Gforth). ;; VARIABLE F F OFF F @ . => 0
BEGIN-STRUCTURE ( "name" -- addr 0 ) Start a structure; the name gives its size once END-STRUCTURE has run.
END-STRUCTURE ( addr n -- ) End a structure and record its size.
+FIELD ( n1 n2 "name" -- n3 ) Define a field of n2 bytes; the name adds its offset to an address.
FIELD: ( n1 "name" -- n2 ) Define a field of one aligned cell.
CFIELD: ( n1 "name" -- n2 ) Define a field of one byte.
FFIELD: ( n1 "name" -- n2 ) Define a field of one float.
DFFIELD: ( n1 "name" -- n2 ) Define a field of one double-precision float.
SFFIELD: ( n1 "name" -- n2 ) Define a field of one single-precision float.

REVERSE ( x1 .. xn n -- xn .. x1 ) Reverse the items. ;; ALSO FUNCTIONAL 1 2 3 3 REVERSE .S PREVIOUS => <3> 3 2 1
TAKE ( x1 .. xm m n -- .. k ) Keep the first n items; k is the new count. ;; ALSO FUNCTIONAL 1 2 3 3 2 TAKE .S PREVIOUS => <3> 1 2 2
DROP-N ( x1 .. xm m n -- .. k ) Drop the first n items; k is the new count. ;; ALSO FUNCTIONAL 1 2 3 3 1 DROP-N .S PREVIOUS => <3> 2 3 2
ZIP ( a1 .. an b1 .. bn n -- a1 b1 .. an bn 2n ) Interleave two lists of n items. ;; ALSO FUNCTIONAL 1 2 3 4 2 ZIP .S PREVIOUS => <5> 1 3 2 4 4
SUM ( x1 .. xn n -- r ) Add the items; 0 for n = 0. ;; ALSO FUNCTIONAL 1 2 3 3 SUM . PREVIOUS => 6
PRODUCT ( x1 .. xn n -- r ) Multiply the items; 1 for n = 0. ;; ALSO FUNCTIONAL 2 3 4 3 PRODUCT . PREVIOUS => 24
MIN-OF ( x1 .. xn n -- r ) The smallest item; n is at least 1. ;; ALSO FUNCTIONAL 3 1 2 3 MIN-OF . PREVIOUS => 1
MAX-OF ( x1 .. xn n -- r ) The largest item; n is at least 1. ;; ALSO FUNCTIONAL 3 1 2 3 MAX-OF . PREVIOUS => 3
MAP-CELLS ( addr n xt -- ) Call xt ( x -- y ) on n cells in place.
FOLD-CELLS ( addr n xt -- r ) Combine n cells with xt ( acc x -- acc' ); the first cell is the first acc, so n is at least 1.
RUNNING-FOLD-CELLS ( addr n xt -- ) Like FOLD-CELLS; cell i becomes the partial result over cells 0 to i.
FILTER-CELLS ( addr n xt -- m ) Move the cells that pass xt ( x -- flag ) to the front; m is how many.
ANY?-CELLS ( addr n xt -- flag ) True if xt ( x -- flag ) is true for some cell; the block stays as it is.
ALL?-CELLS ( addr n xt -- flag ) True if xt ( x -- flag ) is true for every cell.
FIND-CELLS ( addr n xt -- x true | false ) The first cell for which xt ( x -- flag ) is true.
COUNT-IF-CELLS ( addr n xt -- m ) How many cells pass xt ( x -- flag ); the block does not change.
REVERSE-CELLS ( addr n -- ) Reverse n cells in place.
SUM-CELLS ( addr n -- r ) Add n cells; 0 for n = 0.
PRODUCT-CELLS ( addr n -- r ) Multiply n cells; 1 for n = 0.
[DYNAMIC-STRINGS]
$VARIABLE ( "name" -- ) Define a variable that holds an empty string.
$! ( c-addr u a -- ) Store a copy of the text in the string at a.
$@ ( a -- c-addr u ) Fetch the text of the string at a; an empty string is 0 0.
$+! ( c-addr u a -- ) Append text to the string; the text may be part of the string itself.
$@LEN ( a -- u ) The length of the string.
$!LEN ( u a -- ) Set the length; new characters are zero.
$+!LEN ( u a -- c-addr ) Make room for u more characters and give their address.
$INS ( c-addr u a pos -- ) Insert text at offset pos (kept within the string).
(INS) ( c-addr u a pos -- ) The part of $INS that opens the gap; use $INS instead.
$DEL ( a pos u -- ) Delete u characters at pos (kept within the string).
$SPLIT ( c-addr u char -- c-addr u1 c-addr2 u2 ) Split a text at the first char; nothing is allocated. Without char, u1 is u and u2 is 0.
$ITER ( .. a char xt -- .. ) Call xt ( .. c-addr u -- .. ) for each part of the string between the chars.
$INIT ( a -- ) Set the string to empty without freeing it.
$FREE ( a -- ) Free the string and set it to empty.
$TMP ( xt -- c-addr u ) Run xt and give what it printed; valid until the next $TMP.
$EXEC ( xt a -- ) Run xt and append what it printed to the string.
$SLURP ( fid a -- ) Read the rest of a file into the string.
$SLURP-FILE ( c-addr u a -- ) Read a whole file into the string.
$[]VARIABLE ( "name" -- ) Define an empty array of strings.
$[] ( n a -- a' ) The cell of string n; the array grows to hold it.
$[]! ( c-addr u n a -- ) Store a copy of the text as element n.
$[]@ ( n a -- c-addr u ) Fetch element n; past the end it gives 0 0 and the array does not grow.
$[]# ( a -- n ) The number of elements.
$[]+! ( c-addr u a -- ) Add the text as a new last element.
$[]MAP ( a xt -- ) Call xt ( c-addr u -- ) for each element.
$[]SLURP ( fid a -- ) Make the lines of a file the elements; old elements are freed.
$[]SLURP-FILE ( c-addr u a -- ) Make the lines of a named file the elements.
$[]FREE ( a -- ) Free the elements and the array.
[OPERATING-SYSTEM]
SYSTEM ( c-addr u -- ) Run a command in the shell of the system and wait; its exit status is in $?.
SH ( "rest-of-line" -- ) Run the rest of the line as a shell command.
$? ( -- n ) The exit status of the last SYSTEM, SH or CLOSE-PIPE.
OPEN-PIPE ( c-addr u fam -- fileid ior ) Run a command: R/O reads its output, W/O writes to its input.
CLOSE-PIPE ( fileid -- status ior ) Close a pipe and wait for the command.
GETENV ( c-addr u -- c-addr' u' ) The value of an environment variable; u' is 0 if it is not set.
ARGC ( -- a-addr ) Address of the number of arguments, the program name included.
ARGV ( -- a-addr ) Address of a vector of zero-terminated argument strings, as in C.
ARG ( u -- c-addr u' ) Argument u; 0 ARG is the program; 0 0 past the last.
NEXT-ARG ( -- c-addr u ) Take the first argument after the program away and give it; 0 0 if none.
SHIFT-ARGS ( -- ) Take the first argument after the program away.
OPEN-DIR ( c-addr u -- dirid ior ) Open a directory.
READ-DIR ( c-addr u1 dirid -- u2 flag ior ) Read the next name (not . and ..) into the buffer; flag is false at the end.
CLOSE-DIR ( dirid -- ior ) Close a directory.
GET-DIR ( c-addr u -- c-addr u' ) The current directory, in the buffer.
SET-DIR ( c-addr u -- ior ) Change the current directory.
MKDIR-PARENTS ( c-addr u mode -- ior ) Make a directory and the ones above it; one that exists is no error.
BASENAME ( c-addr u -- c-addr' u' ) The file name of a path. ;; ALSO OPERATING-SYSTEM S" /a/b.txt" BASENAME TYPE PREVIOUS => b.txt
DIRNAME ( c-addr u -- c-addr' u' ) The directory of a path, with its last /; empty if none.
FILENAME-MATCH ( c-addr u c-addr2 u2 -- flag ) True if the name matches the pattern (* and ?).
STDIN ( -- fileid ) The standard input as a file.
STDOUT ( -- fileid ) The standard output as a file.
STDERR ( -- fileid ) The standard error as a file.
OUTFILE-EXECUTE ( xt fileid -- ) Run xt with TYPE writing to the file.
INFILE-EXECUTE ( xt fileid -- ) Run xt with KEY and ACCEPT reading from the file.
[BENCHMARK]
BENCH ( xt -- microseconds ) Run xt ( -- ) once and give the time it took.
BENCH-BEST ( n xt -- microseconds ) Run xt ( -- ) n times (at least 1) and give the shortest time.
[TESTING]
T{ ( -- ) Start a test; the stack as it is now stays below the test.
-> ( -- ) The cells since T{ are the actual results.
}T ( -- ) The cells since -> are the expected results; compare, and print and count a difference.
TESTING ( "text" -- ) Name the tests that follow, in the failure lines.
TEST-SUMMARY ( -- ) Print how many tests passed and failed.
SET-EXACT ( -- ) Compare the floats of a test as identical; this is the default. ;; ALSO TESTING T{ 1E 2E F+ -> 3E }T TEST-SUMMARY PREVIOUS => tests: 1 passed, 0 failed
SET-NEAR ( -- ) Compare the floats of a test within REL-NEAR (relatively) or ABS-NEAR (absolutely). ;; ALSO TESTING SET-NEAR T{ 0.1E 0.2E F+ -> 0.3E }T TEST-SUMMARY SET-EXACT PREVIOUS => tests: 1 passed, 0 failed
REL-NEAR ( -- a-addr ) The float variable with the relative tolerance of SET-NEAR; 1E-12 at the start. ;; ALSO TESTING REL-NEAR F@ F. PREVIOUS => 0.000000000001
ABS-NEAR ( -- a-addr ) The float variable with the absolute tolerance of SET-NEAR; 0E at the start. ;; ALSO TESTING ABS-NEAR F@ F. PREVIOUS => 0.
[DEBUG]
TRACE-ON ( -- ) Print every word that runs, with the stack before and after it.
TRACE-OFF ( -- ) Stop the trace.
BREAK ( -- ) Stop the program here for the debugger; without a debugger it does nothing.
